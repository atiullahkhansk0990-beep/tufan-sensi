import os, subprocess

def get_prop(prop):
    try:
        return subprocess.check_output(['getprop', prop]).decode().strip()
    except Exception:
        return "Unknown"

model = get_prop('ro.product.model') or "Android Device"
brand = get_prop('ro.product.brand').upper()
android_ver = get_prop('ro.build.version.release')
density = get_prop('ro.sf.lcd_density') or "440"

mem_kb = 0
try:
    with open('/proc/meminfo') as f:
        for line in f:
            if 'MemTotal' in line:
                mem_kb = int(line.split()[1])
                break
except Exception:
    pass

ram_gb = round(mem_kb / (1024 * 1024), 1) if mem_kb else 4.0
safe_dpi = 420 if ram_gb <= 4.0 else (475 if ram_gb <= 6.0 else 540)
